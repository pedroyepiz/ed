void my_fgets(char string[],int tam)
{
  int i=0 ;
  fflush(stdin);
  fgets(string,tam,stdin);
  while (string[i] != '\0')
  {
    i++;
  }
  string[i-1]='\0';
}

int my_int(int ri, int rf, const char msge[])
{
  int num;
  char my_num[30];
  do{
    printf ("%s",msge);
    my_fgets(my_num,30);
    num = atoi(my_num);
  }while(num < ri || num > rf);
  
  return num;  // REGRESA  EL VALOR YA VALIDADO
}

Tdato gen_nodo(int num)
{
    Tdato nodo=NULL;
    //int num;
    //num = rand()% 98;
    nodo = (Tdato) malloc(sizeof(sTdato));
    if (nodo)
    {
        nodo->dato = num;
        nodo->sig = NULL;
    }    
   return nodo;
}

void servicio (Tdato nodo)
{
    //system ("CLS");
    if (nodo)
    {
        printf ("|%3d|->",nodo->dato);
    }
    else
    {
        printf ("...SIN NODOS...");
    }
   getch();
}

void PUSH(Tdato &pila, Tdato &nodo)
{
     if (nodo)
     {
       nodo->sig = pila;
     }
     pila = nodo;
     nodo = NULL;
}

Tdato POP (Tdato &pila)
{
    Tdato nodo = NULL;
    if (pila)
    {
        nodo = pila;
        pila= nodo->sig;
    }
    return nodo;
}
