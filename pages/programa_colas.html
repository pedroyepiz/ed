typedef int Ted; // Tipo Estructura de Datos,  para ser remplazado y facilitar el servivio 
typedef struct _tnodo{
    Ted dato;
    struct _tnodo *sig;
}sTnodo;
typedef sTnodo *Tnodo;
//****************************************************

Tnodo gen_nodo();
void servicio_basico (Tnodo nodo);
void PUSH(Tnodo &cola, Tnodo &ultimo, Tnodo &nodo);
Tnodo POP(Tnodo &cola);

    menu();

//*********************************
Tnodo gen_nodo(Ted num)
{
    Tnodo nodo=NULL;
    //int num;
    //num = rand()% 98;
    nodo = (Tnodo) malloc(sizeof(sTnodo));
    if (nodo)
    {
        nodo->dato = num;
        nodo->sig = NULL;
    }    
   return nodo;
}
//*****************************************
void servicio_basico (Tnodo nodo)
{
    //system ("CLS");
    if (nodo)
    {
        printf ("|%3d|->",nodo->dato);
    }
    else
    {
        printf ("...SIN NODOS...");
    }
   getch();
}
//***************************************
void PUSH(Tnodo &cola, Tnodo &ultimo, Tnodo &nodo)
{
     if (nodo)
     {
       if (!cola)
       {
          cola = nodo;
          ultimo = nodo;
       } 
       else
       { 
          ultimo->sig = nodo;
          ultimo = nodo;
       }
     }
     nodo = NULL;
}
//******************************************************
Tnodo POP (Tnodo &cola)
{
    Tnodo nodo = NULL;
    if (cola)
    {
        nodo = cola;
        cola= nodo->sig;
    }
    return nodo;
}
//**************************************************
int letrero()
{
    int op;
    system ("CLS");

{
    Tnodo nodo=NULL;

        }

    }while (op != 3);
